function createAsset(THREE, seed, textures, params = {}, addons) {
  // One unit = one metre. Overall closed dimensions follow Walters 51.470.
  // This is an interpretation, not a scan. Construction dimensions are estimates.
  const root = new THREE.Group(); root.name = 'VisoredSallet';
  const steel = new THREE.MeshStandardMaterial({color:0x89939c,metalness:1,roughness:params.roughness ?? 0.32});
  const edgeSteel = new THREE.MeshStandardMaterial({color:0xa2abb2,metalness:1,roughness:0.25});

  const thickness = 0.0018;
  const visor = new THREE.Group(); visor.name = 'VisorPivot';
  visor.position.set(0,0.150,-0.005); root.add(visor);
  function mesh(name, p, indices, uv, parent, material) {
    const g = new THREE.BufferGeometry();
    g.setAttribute('position',new THREE.Float32BufferAttribute(p,3));
    g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
    g.setIndex(indices); g.computeVertexNormals();
    const m = new THREE.Mesh(g,material); m.name=name; parent.add(m); return m;
  }
  // Two actual surfaces joined around their boundary; the underside stays open.
  function plate(name, rows, cols, point, parent, material, skip = () => false) {
    const p=[],uv=[],ix=[];
    for(let side=0;side<2;side++) for(let j=0;j<=rows;j++) for(let i=0;i<=cols;i++) {
      const v=point(i/cols,j/rows,side); p.push(...v); uv.push(i/cols,j/rows);
    }
    const stride=cols+1, layer=(rows+1)*stride;
    for(let s=0;s<2;s++) for(let j=0;j<rows;j++) for(let i=0;i<cols;i++) {
      if(skip(i,j)) continue; const a=s*layer+j*stride+i,b=a+1,c=a+stride,d=c+1;
      if(!s) ix.push(a,c,b,b,c,d); else ix.push(a,b,c,b,d,c);
    }
    const edges=new Map();
    for(let k=0;k<ix.length;k+=3) {
      if(ix[k]>=layer) continue;
      for(let e=0;e<3;e++){const a=ix[k+e],b=ix[k+(e+1)%3],key=Math.min(a,b)+':'+Math.max(a,b);if(edges.has(key))edges.delete(key);else edges.set(key,[a,b]);}
    }
    for(const [a,b] of edges.values())ix.push(b,a,a+layer,b,a+layer,b+layer);
    return mesh(name,p,ix,uv,parent,material);
  }
  const N=48,R=14,p=[],uv=[],ix=[];
  function skull(theta,t,s=0) {
    const c=Math.cos(theta),sn=Math.sin(theta),back=Math.max(-c,0),front=Math.max(c,0);
    const phi=t*Math.PI/2, radius=Math.cos(phi),rise=Math.sin(phi);
    const rim=0.070+0.084*Math.min(1,front/0.32)-0.070*Math.pow(back,3);
    const x=(0.106-s*thickness)*radius*sn;
    const z=(0.128-s*thickness)*radius*c-0.126*Math.pow(back,5)*Math.pow(1-rise,3);
    const keel=0.003*Math.pow(Math.max(0,1-Math.abs(sn)*7),2)*Math.sin(phi*Math.PI);
    return [x,rim+(0.255-rim)*rise+keel-s*thickness,z];
  }
  // Apex is one vertex per surface, avoiding a collapsed ring of zero-area faces.
  for(let s=0;s<2;s++) {
    for(let j=0;j<R;j++) for(let i=0;i<=N;i++) {p.push(...skull(i/N*2*Math.PI,j/R,s));uv.push(i/N,j/R);}
    p.push(0,0.255-s*thickness,0);uv.push(0.5,1);
  }
  const layer=R*(N+1)+1;
  for(let s=0;s<2;s++) {
    for(let j=0;j<R-1;j++) for(let i=0;i<N;i++) {
      const a=s*layer+j*(N+1)+i,b=a+1,c=a+N+1,d=c+1;
      if(!s) ix.push(a,c,b,b,c,d); else ix.push(a,b,c,b,d,c);
    }
    for(let i=0;i<N;i++) {const a=s*layer+(R-1)*(N+1)+i,b=a+1,top=s*layer+layer-1;if(!s)ix.push(a,top,b);else ix.push(a,b,top);}
  }
  for(let i=0;i<N;i++)ix.push(i,i+1,i+layer,i+1,i+1+layer,i+layer);
  const bowl=mesh('Skull_and_integral_tail',p,ix,uv,root,steel);
  // Smooth only duplicated longitude-seam normals; retain separate inner/outer skins.
  const normals=bowl.geometry.attributes.normal,positions=bowl.geometry.attributes.position;
  const same=new Map();
  for(let i=0;i<positions.count;i++){
    const key=[positions.getX(i),positions.getY(i),positions.getZ(i)].map(x=>x.toFixed(7)).join(',');
    if(!same.has(key))same.set(key,[]);same.get(key).push(i);
  }
  function visorPoint(u,v,s=0) {
    const theta=(u-0.5)*Math.PI,c=Math.cos(theta),sn=Math.sin(theta),k=v*7;
    if(k<=4.000001){
      const f=k/4,top=0.143+0.012*Math.pow(Math.abs(sn),8),bottom=0.077+0.040*sn*sn;
      const y=bottom*(1-f)+top*f;
      const z=(0.130+0.018*Math.sin((1-f)*Math.PI*0.85))*c+0.005*Math.pow(c,12)-s*thickness*c;
      return [(0.110-s*thickness)*sn,y-0.150-s*0.00025,z];
    }
    const f=(k-5)/2,low=0.150+0.005*Math.pow(Math.abs(sn),8),high=0.172+0.009*Math.pow(c,8);
    const y=low*(1-f)+high*f;
    const rim=0.070+0.084*Math.min(1,c/0.32),rise=(y-rim)/(0.255-rim),rad=Math.sqrt(Math.max(0,1-rise*rise));
    return [(0.106*rad+0.004-s*thickness)*sn,y-0.150-s*0.00025,(0.128*rad+0.008-s*thickness)*c+0.005];
  }
  plate('Visor_with_open_sight',7,40,visorPoint,visor,steel,(i,j)=>j===4&&Math.abs(((i+0.5)/40-0.5)*Math.PI)<1.18);
  function rivet(name,pos,r,parent,axis) {
    const g=new THREE.SphereGeometry(r,8,4);g.scale(1,1,0.42);
    const o=new THREE.Mesh(g,edgeSteel);o.name=name;o.position.copy(pos);
    o.quaternion.setFromUnitVectors(new THREE.Vector3(0,0,1),axis);parent.add(o);return o;
  }
  for(const sign of [-1,1]) {
    const a=new THREE.Vector3(sign,0,0);
    rivet('Visor_pivot_'+(sign<0?'L':'R'),new THREE.Vector3(sign*0.110,0,0.003),0.009,visor,a);
    rivet('Pivot_pin_'+sign,new THREE.Vector3(sign*0.114,0,0.003),0.0035,visor,a);
  }
  // Sparse liner rivets on the shell, rather than decorative fantasy studs.
  for(let i=0;i<11;i++) {
    const th=1.72+i*(2*Math.PI-3.44)/10;
    const v=new THREE.Vector3(...skull(th,0.09,0));
    const a=new THREE.Vector3(Math.sin(th),0.20,Math.cos(th)).normalize();
    v.addScaledVector(a,0.0008);rivet('Lining_rivet_'+i,v,0.0028,root,a);
  }
  // Rolled lower visor edge is an eight-sided sweep with a deliberate 1.4mm radius.
  const path=[];
  for(let i=0;i<=32;i++)path.push(new THREE.Vector3(...visorPoint(i/32,0,0)));
  const curve=new THREE.CatmullRomCurve3(path);
  const roll=new THREE.Mesh(new THREE.TubeGeometry(curve,32,0.0014,6,false),edgeSteel);
  roll.name='Visor_rolled_edge';visor.add(roll);
  // Each plate is a closed thin solid. Orient by signed volume, not double-sided shading.
  root.traverse(o=>{
    if(!o.isMesh)return;
    const g=o.geometry,a=g.attributes.position,idx=g.index;let vol=0;
    if(idx){for(let i=0;i<idx.count;i+=3){const A=new THREE.Vector3().fromBufferAttribute(a,idx.getX(i)),B=new THREE.Vector3().fromBufferAttribute(a,idx.getX(i+1)),C=new THREE.Vector3().fromBufferAttribute(a,idx.getX(i+2));vol+=A.dot(B.cross(C));}
      if(vol<0){for(let i=0;i<idx.count;i+=3){const b=idx.getX(i+1);idx.setX(i+1,idx.getX(i+2));idx.setX(i+2,b);}g.computeVertexNormals();}}
  });
  for(const group of same.values())if(group.length>1){const n=new THREE.Vector3();for(const i of group)n.add(new THREE.Vector3().fromBufferAttribute(normals,i));n.normalize();for(const i of group)normals.setXYZ(i,n.x,n.y,n.z);}
  // Normalize the CLOSED assembly to documented outer extents, baking geometry
  // and translation so the root remains unit scale. Open pose naturally differs.
  root.updateMatrixWorld(true);
  const bounds=new THREE.Box3().setFromObject(root),size=bounds.getSize(new THREE.Vector3());
  const f=new THREE.Vector3((params.widthMm??225)/1000/size.x,(params.heightMm??255)/1000/size.y,(params.lengthMm??396)/1000/size.z);
  root.traverse(o=>{if(o!==root)o.position.multiply(f);if(o.isMesh){
    // Bake pre-existing rotations first (rivet normals).
    o.geometry.applyQuaternion(o.quaternion);o.quaternion.identity();o.geometry.scale(f.x,f.y,f.z);
  }});
  visor.rotation.x=-THREE.MathUtils.degToRad(params.visorAngle??0);
  root.userData={units:'metres',reference:'Walters Art Museum 51.470, 1480-1490',referenceDimensionsMm:{height:255,width:225,length:396},construction:'Interpretation; plate thickness 1.8mm and 7mm sight opening are estimates',forward:'+Z',up:'+Y',pivot:'VisorPivot, local X axis',source:'https://art.thewalters.org/object/51.470/'};
  return {root,dispose:()=>{root.traverse(o=>{if(o.isMesh)o.geometry.dispose();});[steel,edgeSteel].forEach(m=>m.dispose());}};
}
